\pgfplotstablecreatecol[copy column from table={\heapsortdata}{t_heapsort}]
  {t_heapsort}{\loglindata}
\pgfplotstablecreatecol[copy column from table={\heapsortpeordata}{t_heapsort_peor}]
  {t_heapsort_peor}{\loglindata}
\pgfplotstablecreatecol[copy column from table={\quicksortdata}{t_quicksort}]
  {t_quicksort}{\loglindata}

\sisetup{group-separator={\ }, group-minimum-digits=4}
